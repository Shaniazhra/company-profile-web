
===== app/Http/Controllers/Admin/SiteSettingController.php =====
<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\SiteSetting;
use App\Support\AdminForm;
use Illuminate\Http\Request;

class SiteSettingController extends Controller
{
    /** Harus sama dengan $translatable di model SiteSetting. */
    private array $translatable = ['tagline', 'heart_statement', 'impact_statement', 'about_text'];

    public function edit()
    {
        return view('admin.site.edit', [
            'site' => SiteSetting::first() ?? new SiteSetting,
        ]);
    }

    public function update(Request $request)
    {
        $data = $request->validate([
            'site_name' => ['required', 'string', 'max:120'],
            'logo' => AdminForm::imageRules(),
            'hero_image' => ['nullable', 'image', 'mimes:png,jpg,jpeg,webp', 'max:5120'],
            ...AdminForm::translationRules('tagline', max: 200),
            ...AdminForm::translationRules('heart_statement'),
            ...AdminForm::translationRules('impact_statement'),
            ...AdminForm::translationRules('about_text', max: 5000),
        ]);

        $site = SiteSetting::first() ?? new SiteSetting;

        $site->site_name = $data['site_name'];

        foreach ($this->translatable as $field) {
            $site->setTranslations($field, AdminForm::clean($data[$field] ?? []));
        }

        $site->logo = AdminForm::syncImage($request, $site->logo, 'logo', 'site');
        $site->hero_image = AdminForm::syncImage($request, $site->hero_image, 'hero_image', 'site');
        $site->save();

        return back()->with('status', 'Pengaturan situs disimpan.');
    }
}

===== app/Models/SiteSetting.php =====
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Attributes\Fillable;
use Illuminate\Database\Eloquent\Model;
use Spatie\Translatable\HasTranslations;

#[Fillable([
    'site_name', 'logo', 'favicon',
    'tagline', 'heart_statement', 'impact_statement', 'about_text',
    'email', 'instagram_url', 'linkedin_url',
])]
class SiteSetting extends Model
{
    use HasTranslations;

    public array $translatable = ['tagline', 'heart_statement', 'impact_statement', 'about_text'];

    private static ?self $instance = null;

    /** Tabel ini hanya punya satu baris. Kalau belum ada, dibuat otomatis. */
    public static function current(): self
    {
        return self::$instance ??= static::query()->first()
            ?? static::create(['site_name' => 'Dream Ratio']);
    }

    public function logoUrl(): ?string
    {
        return $this->logo ? asset('storage/' . $this->logo) : null;
    }

    public function faviconUrl(): ?string
    {
        return $this->favicon ? asset('storage/' . $this->favicon) : null;
    }
}
